var FLAGparam = 4;
var FLAGbackGround = 2;

var eps = 0.0001;

var Black = [0,0,0];                                   // FLAGbackGround = 0
var NDVI = index(B08, B04);                             // FLAGbackGround = 1
var NDWI = index(B03, B08);
var TrueColor = [B04*2.5, B03*2.5, B02*2.5];             // FLAGbackGround = 2

// Se2WaQ parameters (safe divisions)
var Chl_a = 4.26 * Math.pow(B03/(B01+eps), 3.94);        
var Cya   = 115530.31 * Math.pow((B03 * B04)/(B02+eps), 2.38);
var Turb  = 8.93 * (B03/(B01+eps)) - 6.39;
var CDOM  = 537 * Math.exp(-2.93 * B03/(B04+eps));
var DOC   = 432 * Math.exp(-2.24 * B03/(B04+eps));
var Color = 25366 * Math.exp(-4.53 * B03/(B04+eps));

// Ranges aligned with your GEE visualization
var scaleChl_a = [0,10,20,30,40,50];
var scaleCya   = [0,20,40,60,80,100];
var scaleTurb  = [0,4,8,12,16,20];
var scaleCDOM  = [0,1,2,3,4,5];
var scaleDOC   = [0,8,16,24,32,40];
var scaleColor = [0,10,20,30,40,50];

// Same palette used in GEE
var s = 255;
var colorScale = [
  [73/s,111/s,242/s],
  [130/s,211/s,95/s],
  [254/s,253/s,5/s],
  [253/s,0/s,4/s],
  [142/s,32/s,38/s],
  [217/s,124/s,245/s]
];

if (NDWI<0) {
  if ( FLAGbackGround == 0 ) {
    return Black;
  } else if ( FLAGbackGround == 1 ) {
    return [0, .5*(NDVI+1), 0];
  } else if ( FLAGbackGround == 2 ) {
    return TrueColor;
  }
} else {
  switch ( FLAGparam ) {
    case 0:
     return colorBlend(Chl_a, scaleChl_a, colorScale);
     break;
    case 1:
      return colorBlend(Cya, scaleCya, colorScale);
      break;
    case 2:
      return colorBlend(Turb, scaleTurb, colorScale);
      break;
    case 3:
      return colorBlend(CDOM, scaleCDOM, colorScale);
      break;
    case 4:
      return colorBlend(DOC, scaleDOC, colorScale);
      break;
    case 5:
      return colorBlend(Color, scaleColor, colorScale);
      break;
    default:
      return TrueColor;
  }
}
